library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL; --Pour opéeration integer

entity pi is
    generic (
        k_coeff: integer range -32768 to 32768-1 := 1; --Coefficient K dans l'équation de récurrence
        coeff_decalage: integer range 0 to 15 := 0; --Coefficient de décalage post opération
        clipping_low: integer range -32768 to 32768-1 := -6249; --Valeur minimale de sortie
        clipping_high: integer range -32768 to 32768-1 := 6249; --Valeur minimale de sortie
        clk_frequency: integer range 0 to 125000000 := 5000); --Valeur de la fréquence d'échzntillonage
    Port ( reset : in STD_LOGIC; --Signal de reset du système
           clk_125 : in STD_LOGIC; --Horloge de cadencage du système
           target : in STD_LOGIC_VECTOR (15 downto 0); --Signal d'entrée de référence
           feedback: in STD_LOGIC_VECTOR(15 downto 0); --Signal d'entrée de feedback
           y_out : out STD_LOGIC_VECTOR (15 downto 0)); -- Vecteur de sortie
end pi;

architecture Behavioral of pi is
    signal epsilon: std_logic_vector(15 downto 0) :=  (others => '0'); --Correspondant à espilon
    signal sortie_sommateur: std_logic_vector(47 downto 0) := (others => '0'); --Signal de sortie du multiplicateur DSPs
    signal compteur : integer := 0; --Compteur pour la synchronissité
    signal clk_int: std_logic := '0';  --Horloge interne
    signal y_out_sign: signed (47 downto 0) := (others => '0'); --Interne signé pour le décalage
    --Composant DSP
    COMPONENT dsp_macro_0 IS
    PORT (
        CLK : IN STD_LOGIC;
        A : IN STD_LOGIC_VECTOR(15 DOWNTO 0);
        B : IN STD_LOGIC_VECTOR(15 DOWNTO 0);
        P : OUT STD_LOGIC_VECTOR(47 DOWNTO 0)
  );
    END COMPONENT;
begin
    -- Composant
    sommateur_pi: dsp_macro_0
    port map(A => epsilon, B => std_logic_vector(to_signed(k_coeff, 16)), CLK => clk_int, P => sortie_sommateur); --Mapping
    horloge: process(clk_125) --Process horloge
    begin
        if(rising_edge(clk_125)) then --Si front montant horloge
            if (reset = '1') then --Si reset actif
                compteur <= 0; --RAZ compteur
                clk_int <= '0'; --Horloge bas
            else
                if(compteur = 125000000/clk_frequency -1) then --Si le compteur est à la valeur maximale
                    compteur <= 0; --RAZ compteur
                    clk_int <= '1'; --Horloge haut
                else --Sinon
                    compteur <= compteur + 1; --Incrémentation compteur
                    clk_int <= '0'; --Horloge bas
                end if;
            end if;
        end if;
    end process;
    
    --Combinatoire de sortie
    epsilon <= std_logic_vector(to_signed(to_integer(signed(target)) - to_integer(signed(feedback)), 16)); --A tout instant, espilon vaut différence entre feedback et consigne
    y_out_sign <= (others => '0') when (reset = '1') else shift_right(signed(sortie_sommateur), coeff_decalage); --Récupérationd e la valeur shiftée
    y_out <= std_logic_vector(y_out_sign(15 downto 0)) when ((clipping_low <= y_out_sign) and (clipping_high >= y_out_sign)) else std_logic_vector(to_signed(clipping_low, 16)) when (to_integer(y_out_sign) < clipping_low) else std_logic_vector(to_signed(clipping_high, 16)) ; --Affectation sortie en valeur absolue & clipping
end Behavioral;
